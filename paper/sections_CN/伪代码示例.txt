Algorithm 2: path crossover operator

Input: two selected feasible paths: path_p and path_q
Output: two crossed feasible paths: path'_p and path'_q

1 let path_p = [S, P_1, ..., P_n, T], path_q = [S, P^*_1, ..., P^*_m, T];
2 for i = 1 to n do
3     select an intermediate point P_i of path_p;
4     find the closest intermediate point P^*_j of path_q to P_i;
5     dis_i = distance(P_i, P^*_j);
6     Dis(i) <- record dis_i and the corresponding points (P_i, P^*_j);
7 end
8 Dis_sort <- Sort the elements of Dis in ascending order based on the value of dis;
9 for index = 1 to n do
10    (P_i_var, P^*_j_var) <- find a pair of points corresponding to Dis_sort(index);
11    if CollisionFree(P_i_var, P^*_j_var) then
12        path'_p = [S, P_1, ..., P_i_var, P^*_j_var, P^*_j_var+1, ..., P^*_m, T];
13        path'_q = [S, P^*_1, ..., P^*_j_var, P_i_var, P_i_var+1, ..., P_n, T];
14        if ObjectiveValue(path'_p) >= ObjectiveValue(path_p) ||
15           ObjectiveValue(path'_q) >= ObjectiveValue(path_q) then
16            break;
17        end
18    end
19    if index == n then
20        path'_p = path_p, and path'_q = path_q;
21    end
22 end
